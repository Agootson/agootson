async function saveBlogPost(event) { event.preventDefault(); const title = document.getElementById("blogTitle").value.trim(); const category = document.getElementById("blogCategory").value.trim(); const tags = document.getElementById("blogTags").value .split(",") .map(tag => tag.trim()) .filter(Boolean); const excerpt = document.getElementById("blogExcerpt").value.trim(); // Correct database column: cover_image const cover_image = document.getElementById("blogCover").value.trim() || null; const content = document.getElementById("blogContent").value.trim(); const status = document.getElementById("blogStatus").value; const dateValue = document.getElementById("blogDate").value; const editId = document.getElementById("blogEditId").value; if (!title || !category || !excerpt || !content) { showNotice("Complete all required article fields.", "error"); return; } const slugBase = blogSlugify(title); if (!slugBase) { showNotice("Enter a valid article title.", "error"); return; } let slug = slugBase; if (!editId) { slug = `${slugBase}-${Date.now().toString(36)}`; } if (status === "scheduled" && !dateValue) { showNotice("Choose a publication date.", "error"); return; } let dateIso = null; if (dateValue) { const parsed = new Date(dateValue); if (Number.isNaN(parsed.getTime())) { showNotice("The publication date is invalid.", "error"); return; } dateIso = parsed.toISOString(); } const { data: { user } } = await supabaseClient.auth.getUser(); currentBlogUser = user || currentBlogUser; const payload = { title, slug, excerpt, content, cover_image, category, tags, status, published_at: status === "published" ? (dateIso || new Date().toISOString()) : null, scheduled_at: status === "scheduled" ? dateIso : null, author_id: currentBlogUser?.id || null }; if (editId) { const existing = blogPosts.find(post => post.id === editId); if (existing) { payload.slug = existing.slug; payload.author_id = existing.author_id || currentBlogUser?.id || null; } } const saveButton = document.getElementById("saveBlogPost"); saveButton.disabled = true; setStatus( document.getElementById("blogEditorStatus"), "Saving article to Supabase..." ); try { let result; if (editId) { result = await supabaseClient .from("blog_posts") .update(payload) .eq("id", editId) .select() .single(); } else { result = await supabaseClient .from("blog_posts") .insert(payload) .select() .single(); } if (result.error) throw result.error; showNotice( status === "published" ? "Article published successfully!" : status === "scheduled" ? "Article scheduled successfully!" : "Draft saved successfully!", "success" ); document.getElementById("blogEditId").value = ""; document.getElementById("blogEditor").reset(); await loadBlogPosts(); } catch (error) { console.error("Article save error:", error); showNotice( "Could not save article: " + error.message, "error" ); } finally { saveButton.disabled = false; saveButton.innerHTML = ' Publish Article'; } }
